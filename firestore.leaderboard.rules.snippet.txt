// Shared profiles + Gmail verify/claim — paste INSIDE match /databases/{database}/documents { ... }
// Email sending: coach-api with RESEND_API_KEY or SENDGRID_API_KEY (Render free blocks SMTP).

rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {

    // Owner: full access to their user tree
    match /users/{userId}/{document=**} {
      allow read, write: if request.auth != null && request.auth.uid == userId;
    }

    // Guest: read shared athlete data after email verify
    match /users/{userId}/athletes/{athleteId}/{document=**} {
      allow read: if request.auth != null
        && request.auth.token.email != null
        && exists(/databases/$(database)/documents/profileLinks/$(request.auth.token.email.lower()))
        && get(/databases/$(database)/documents/profileLinks/$(request.auth.token.email.lower())).data.hostUid == userId
        && get(/databases/$(database)/documents/profileLinks/$(request.auth.token.email.lower())).data.profileId == athleteId
        && get(/databases/$(database)/documents/profileLinks/$(request.auth.token.email.lower())).data.verified == true;
    }

    match /emailIndex/{emailKey} {
      allow read: if request.auth != null;
      allow create, update: if request.auth != null
        && request.auth.token.email != null
        && request.auth.token.email.lower() == emailKey
        && request.resource.data.uid == request.auth.uid;
      allow delete: if false;
    }

    match /profileLinks/{emailKey} {
      allow create: if request.auth != null
        && request.resource.data.hostUid == request.auth.uid;

      allow update: if request.auth != null && (
        resource.data.hostUid == request.auth.uid
        || (
          request.auth.token.email != null
          && request.auth.token.email.lower() == emailKey
          && request.resource.data.token == resource.data.token
        )
      );

      allow read, delete: if request.auth != null
        && (resource.data.hostUid == request.auth.uid
          || (request.auth.token.email != null
            && request.auth.token.email.lower() == emailKey));
    }
  }
}
